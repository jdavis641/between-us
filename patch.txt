+      if (!session) {
+        throw new Error("Auth Rejected: Session expired or missing.");
+      }
+      
+      const res = await fetch`;
+
+  code = code.replace(/    try \{\s*const res = await fetch/g, sessionCheck);
+
+  const tokenHeader = `        headers: {
